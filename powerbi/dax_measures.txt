This file contains all DAX (Data Analysis Expressions) formulas required to build 
the "EMPLOYEE ATTRITION ANALYSIS" Power BI Dashboard.

These formulas are designed for the cleaned dataset table: 'EMPLOYEE_ATTRITION' 
(imported from dataset/employee_attrition_cleaned.csv).


-----------------------------------------------------------------------------
1. FIVE CORE KPI CARD MEASURES (Matches Dashboard Top Cards)
-----------------------------------------------------------------------------

-- KPI 1: Total Employees
Total Employees = 
COUNTROWS('EMPLOYEE_ATTRITION')

-- KPI 2: Active Employees (Retained Workforce)
Active Employees = 
CALCULATE(
    COUNTROWS('EMPLOYEE_ATTRITION'),
    'EMPLOYEE_ATTRITION'[Attrition] = "No"
)

-- KPI 3: Employees Who Left (Voluntary Exits)
Employees Who Left = 
CALCULATE(
    COUNTROWS('EMPLOYEE_ATTRITION'),
    'EMPLOYEE_ATTRITION'[Attrition] = "Yes"
)

-- KPI 4: Overall Attrition Rate (%)
Overall Attrition Rate % = 
DIVIDE([Employees Who Left], [Total Employees], 0) * 100

-- KPI 5: Average Employee Age
Average Employee Age = 
ROUND(AVERAGE('EMPLOYEE_ATTRITION'[Age]), 1)


-----------------------------------------------------------------------------
2. COMPLEMENTARY METRICS & AVERAGES
-----------------------------------------------------------------------------

Retention Rate % = 
100 - [Overall Attrition Rate %]

Average Monthly Salary = 
AVERAGE('EMPLOYEE_ATTRITION'[MonthlyIncome])

Average Years at Company = 
AVERAGE('EMPLOYEE_ATTRITION'[YearsAtCompany])


-----------------------------------------------------------------------------
3. CONDITIONAL & SEGMENTED RISK MEASURES (For Drilldowns & Tooltips)
-----------------------------------------------------------------------------

-- OverTime Attrition Rate %
OverTime Attrition Rate % = 
CALCULATE(
    [Overall Attrition Rate %],
    'EMPLOYEE_ATTRITION'[OverTime] = "Yes"
)

-- Non-OverTime Attrition Rate %
Non-OverTime Attrition Rate % = 
CALCULATE(
    [Overall Attrition Rate %],
    'EMPLOYEE_ATTRITION'[OverTime] = "No"
)

-- Early Career Flight Risk (0-2 Years Tenure)
Early Career Attrition Rate % = 
CALCULATE(
    [Overall Attrition Rate %],
    'EMPLOYEE_ATTRITION'[TenureBand] = "0-2 Years"
)

-- Low Salary Flight Risk (<$3,500/Month)
Low Salary Attrition Rate % = 
CALCULATE(
    [Overall Attrition Rate %],
    'EMPLOYEE_ATTRITION'[SalaryBand] = "Low (<3.5k)"
)

-- Dynamic Health Status Indicator
Attrition Status Indicator = 
VAR Rate = [Overall Attrition Rate %]
RETURN
    SWITCH(
        TRUE(),
        Rate >= 20.0, "CRITICAL RISK (RED)",
        Rate >= 15.0, "ELEVATED RISK (AMBER)",
        "STABLE / HEALTHY (GREEN)"
    )
